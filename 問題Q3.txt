//約據-刪除保人
function deleteSurety(obj) {
    $(obj).closest(`.identities`).nextAll().each(function () {
        let $creThis = $(this);
        let callLoanID = $creThis.closest(`.identities`).attr("id"); //identities2,identities3...
        let callLoanIDNum = callLoanID.substring(callLoanID.length - 1);//2,3...
        if (callLoanID.length > 11) {//12位
            callLoanIDNum = callLoanID.substring(callLoanID.length - 2);
        }
        let numNew = parseInt(callLoanIDNum) - 1;//字串轉數字

        $("#" + callLoanID).find(".addNum").text(numNew + `：預計動用日期：`);//換編號
        $("#" + callLoanID).find(".accCallNumber").val(numNew);
        callLoanID = callLoanID.replace(callLoanID, "identities" + numNew); //將舊的identitiesX替代掉
        $creThis.closest(`.identities`).attr("id", callLoanID)
    })
    $(obj).closest(`.identities`).remove();

    //綜合契約書連動
    guaAmountDel()// 刪除連帶保證人金額欄位
    responDel() // 刪除負責人欄位
}
