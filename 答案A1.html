使用 jQuery 監聽 keydown 事件 $('.phoneTime').on('keydown', function (e) { if (e.which === 13 || e.key === "Enter") { e.preventDefault(); // 防止表單被送出 let $input = $(this); let val = $input.val().trim(); // 判斷是否為剛好 4 碼純數字 (例如 "0930", "1420") if (/^\d{4}$/.test(val)) { let hour = parseInt(val.substring(0, 2), 10); let minute = parseInt(val.substring(2, 4), 10); // 驗證時間範圍：小時 00~23，分鐘 00~59 if (hour >= 0 && hour <= 23 && minute >= 0 && minute <= 59) { let formattedTime = `${val.substring(0, 2)}:${val.substring(2, 4)}`; // 用 jQuery 寫入格式化後的字串，並觸發 change 事件 $input.val(formattedTime).trigger('change').blur(); } else { alert("請輸入有效的時間範圍（小時 00~23，分鐘 00~59）"); $input.val(""); } } } });
