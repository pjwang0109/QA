            uintData["vid"] = $unit.find("input.vid").val() || "";
            // 統一編號
            uintData["item"] = $unit.find("input.item").val() || "";
            // 進貨廠商名稱

            // 佔進貨比重
            uintData["rratio"] = $unit.find("input.rratio").val() || "";
            // 國別
            uintData["nation"] = $unit.find("select.nation").val() || "";
            // 付款條件
            uintData["pterms"] = $unit.find("input.pterms").val() || "";

            const chkArr = ["vid", "item", "vname", "rratio", "nation", "pterms"];
