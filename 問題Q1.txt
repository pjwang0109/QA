function addCollDiv() {
    let userObj = getUidToken();
    let uid = userObj.uid;
    let token = userObj.token;
    let business = $('.business').val();

    if ($('#collModel form').length != 0) {
        collModalinit();
    } else {
        let url = ajaxhttpIPHostHtml('/acceptcase/ademo_AC_collModal.html');

        //擔保品選單
        let s020010008 = getS020010008(uid, token, business);

        $.when($.ajax(url), $.ajax(s020010008)).done(function (html, seldata) {
            $('#modalAll').append(html[0]);
            let option = '<option value="default">請選擇...</option>';
            $('#collModel .selCollItem_mod select').empty().append(option);
            $('#msg_token').text(seldata[0].token);
            if (seldata[0].rc == "M0000") {
                let select = seldata[0].result;
                $.each(select, function (item, jdata) {
                    $('#collModel .selCollItem_mod select').append(option.replace(/default/g, jdata.id).replace(/請選擇.../g, jdata.mean));
                });
            } else {
                apiErrorHandler(seldata[0]);
            }
            collModalinit();

            $('#appraisalcode_colMod').on({
                keyup: function (e) {
                    var flag = e.target.isNeedPrevent;
                    if (flag) return;
                    let val = $(this).val();
                    //只留數字
                    //digitalboxHandler(this)
                    checkappraisalcode(val);
                    e.target.keyEvent = false;

                },
                keydown: function (e) {
                    e.target.keyEvent = true;
                },
                input: function (e) {
                    console.log(e.target.keyEvent);
                    if (!e.target.keyEvent) {
                        let val = $(this).val();
                        checkappraisalcode(val);
                    }
                },
                compositionstart: function (e) {
                    e.target.isNeedPrevent = true;
                },
                compositionend: function (e) {
                    e.target.isNeedPrevent = false;

                }
            });
        });
    }
