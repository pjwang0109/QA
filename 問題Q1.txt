function openCRRRatingmodalWindow(obj) {
    $(obj).prop("disabled", true);//先鎖住
    loadingShow("資料處理中...");
    let userObj = getUidToken();
    let uid = userObj.uid;
    let token = userObj.token;
    let ufid = $('#creditRatingTittle').attr("ufid");
    let dataObj = { "uid": uid, "token": token, "ufid": ufid };
    console.log(dataObj)
    sessionStorage.setItem("openCRRRatingWindowData", JSON.stringify(dataObj));

    setTimeout(() => {
        window.open("../../acceptcase/page/oldCase_CR.html", undefined, "menubar=no");
        $(obj).prop("disabled", false);
    }, 500)
    loadingHide();
}
