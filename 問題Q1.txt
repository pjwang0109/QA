function icCreditAgreementFill($card, val) {
    icSetPerson($card, val.id);
    icSetDate($card, val);
    $card.find(".seal2").first().val(val.seal2 || "");
}

//1授信約定書-暫存取值（S060001037 的 data）
function icCreditAgreementCollect($card) {
    const date = icGetDate($card);
    return {
        "id": icGetPerson($card),
        "year": date.year,
        "month": date.month,
        "day": date.day,
        "seal2": $card.find(".seal2").first().val() || ""
    };
}
