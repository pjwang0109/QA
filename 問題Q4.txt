function icJointGuaranteeFixedTermCollect($card) {
const guarantors = [];
    $card.find(".guaranteePersonList .icPersonGroup").each(function (idx) {
        const personId = icGetPerson($card, idx);
        if (personId) {
            guarantors.push(personId);
        }
    });

    const date = icGetDate($card);

    return {
        "id": icGetPerson($card),
        "year": date.year,
        "month": date.month,
        "day": date.day,
        "branch": $card.find(".guaranteeBranch").val() || "",
        "seal2": $card.find(".seal2").first().val() || ""
    };
}


//人員-取值：回傳所選身分別的統編/ID，先依頁面現況刷新一次，確保是最新的
function icGetPerson($card, groupIndex) {
    const $group = $card.find(".icPersonGroup").eq(groupIndex || 0);
    icRefreshPersonGroup($group, icIdentityList());
    return $group.data("icPersonId") || "";
}
