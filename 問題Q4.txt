/**
 * 依選取的連帶保證人選項，從 .N5 #GCC 取得對應的擔保額度（gccMoney）
 * 優先用「顯示名稱（如 連帶保證人2）+ 統編」雙重精準比對，避免同統編重複出現時抓錯筆
 * @param {Object} optionItem icOptions 裡的物件 { id, label, name, ... }
 * @returns {string} 擔保額度（含千分位字串，無資料回傳空字串）
 */
function icIdentityGuaranteeMoney(optionItem) {
    if (!optionItem) { return ""; }
    let money = "";
    const targetLabel = $.trim(optionItem.label || "");
    const targetId = $.trim(optionItem.id || "");

    // 1. 優先用「角色名稱＋序號」與「統編」雙重比對
    $(".N5 #GCC .guarantor-block").each(function () {
        const $block = $(this);
        const blockId = $.trim($block.attr("data-id") || $block.find("input.code").val() || "");
        const labelText = $.trim($block.find(".jsGuarantor").first().text()).replace(/[：:]$/, "");

        if (blockId === targetId && labelText === targetLabel) {
            money = $.trim($block.find("input.gccMoney").first().val() || "");
            return false;
        }
    });

    // 2. 防呆備援：若標題文字格式不同沒對到，退回用統編抓第一筆符合的
    if (money === "" && targetId) {
        $(".N5 #GCC .guarantor-block").each(function () {
            const $block = $(this);
            const blockId = $.trim($block.attr("data-id") || $block.find("input.code").val() || "");
            if (blockId === targetId) {
                money = $.trim($block.find("input.gccMoney").first().val() || "");
                return false;
            }
        });
    }

    return money;
}

//刷新單張卡片的「連帶保證人擔保額度」（永遠只認第一筆保證人）
function icRefreshGuAmount($card) {
    const $firstSelect = $card.find(".guaranteePersonList .icPersonGroup").first().find(".icPerson");
    const selectEl = $firstSelect[0];
    const options = $firstSelect.data("icOptions") || [];
    const selectedItem = (selectEl && selectEl.selectedIndex > 0) ? options[selectEl.selectedIndex - 1] : null;

    $card.find(".icGuPersonAmount").val(icIdentityGuaranteeMoney(selectedItem));
}

//事件委派綁定（檔案載入時執行一次，負責動態連動）
function icBindGuFixedTermEvents() {
    const cardSelector = "#indivdualCtableDiv .indivdualC_TableList[data-ictype='JOINT_GUARANTEE_FIXED_TERM']";

    // 1. 切換「第一筆」立連帶保證書人時，更新擔保額度（第二筆以後變更不影響額度）
    $(document).on("change", `${cardSelector} .guaranteePersonList .icPersonGroup:first-child .icPerson`, function () {
        icRefreshGuAmount($(this).closest(".indivdualC_TableList"));
    });

    // 2. 上方 .N5 #GCC 的擔保額度（.gccMoney）修改或千分位格式化後，同步更新卡片額度
    $(document).on("input blur change", ".N5 #GCC input.gccMoney", function () {
        $(cardSelector).each(function () {
            icRefreshGuAmount($(this));
        });
    });

    // 3. 上方身分別區塊異動（導致連帶保證人1、2重新編號或刪除）時，延遲 120ms 待共用選單重繪後同步額度
    $(document).on("input change", "#genderDiv input, #genderDiv select", function () {
        setTimeout(function () {
            $(cardSelector).each(function () {
                icRefreshGuAmount($(this));
            });
        }, 120);
    });

    // 4. 上方主畫面「帳號」改變時，同步更新所有卡片的 .icGuAccountNo
    // ⚠️ 請將 ".accountNo" 替換為主畫面上帳號 input 的實際選取器
    $(document).on("input change", ".accountNo", function () {
        const accountNo = $(this).val() || "";
        $(`${cardSelector} .icGuAccountNo`).val(accountNo);
    });
}
icBindGuFixedTermEvents();

//本票及授權書-回填（val 為下行的 data）
function icCashiersFill($card, val) {
    const num = $card.attr("data-num");
    let $identitiesAllDiv = $card.find(".identitiesAllDiv");

    //發票人或共同發票人開始
    $.each(val.identities, function (i, v) {
        if (i > 0) {
            //判斷如果大於一筆時
            i++;
            let $div = `<div class="mt-2 identitiesSelectDiv${i}">
            <input type="hidden" class="identitiesNo" name="no" value="${v.code}">
            <select class="form-control mr-2 identitiesRoleSelect">
                <option value="">請選擇</option>
                <option value="1">發票人</option>
                <option value="2">共同發票人</option>
            </select>
            <select class="form-control mr-2 identitiesSelect">
                <option value="">請選擇</option>
            </select>
            <button type="button" class="btn btn-danger mr-2" onclick="delIdentitiescBtn(this)">×</button>
            </div>`;
            $identitiesAllDiv.append($div);
            identitiesSelect(num, i); //取出ID並塞入選單方法
            $identitiesAllDiv.find(`.identitiesSelectDiv${i}`).find(".identitiesRoleSelect").val(v.type);
            $identitiesAllDiv.find(`.identitiesSelectDiv${i}`).find(".identitiesSelect").val(v.name);
        } else {
            //第一筆
            $identitiesAllDiv.find(`.identitiesSelectDiv1`).find(".identitiesRoleSelect").val(v.type);
            $identitiesAllDiv.find(`.identitiesSelectDiv1`).find(".identitiesSelect").val(v.name);
            $identitiesAllDiv.find(`.identitiesSelectDiv1`).find(".identitiesNo").val(v.code);
        }
    });
    //發票人或共同發票人結束

    let quota_Class = $card.find(".quota");
    //本票及授權書-選擇額度
    $.each(val.quota, function (item, quotaVal) {
        $.each(quota_Class, function (item, quota) {
            let screenVal = $(quota).val();
            if (screenVal == quotaVal) {
                $(quota).prop("checked", true);
            }
        })
    })
    //本票及授權書-幣別
    let $quotaChecked = $card.find(".quota:checked");
    $.each($quotaChecked, function (i, input) {
        let currency = input.getAttribute("currency");
        $card.find(".indivdualCurrency").val(currency);
    });
    $card.find(".amount").val(val.amount);//幣別金額
    $card.find("#rateType").val(val.type);//利率型式
    //本票及授權書-延遲利息計收費方式勾選項目
    let mode_Class = $card.find(".delay_interest");
    $.each(val.mode, function (item, modeVal) {
        $.each(mode_Class, function (item, mode) {
            let screenVal = $(mode).val();
            if (screenVal == modeVal) {
                $(mode).prop("checked", true);
            }
        })
    })
    $card.find(".interest").val(val.interest);//延遲利息%
    $card.find(".other").val(val.other);//其他
    $card.find(".seal2").val(val.seal2);//授權書簽章
}

//本票及授權書-暫存取值（S060001037 的 data）
//由 saveOpJSON 原本的本票取值搬來，欄位順序 quota, amount, type, mode, interest, other, seal2, identities 不可變動（S060000005 轉回 cashiers 時沿用此順序）
//data 內 quota:額度Type, amount:金額, type:利率型式, mode:延遲利息計收方式(複選), interest:加碼年息, other:其他, seal2:授權書簽章, identities:發票人
function icCashiersCollect($card) {
    const classIndividualArr = [".quota", ".amount", ".type", ".delay_interest", ".interest", ".other", ".interest", ".seal2", ".identities"];
    const $thisbody = $card.find('.card-body');
    const individualObj = new Object;
    $.each(classIndividualArr, function (item, classIndivdual) {
        const dataIndividual = classIndivdual.replace('.', '');
        const classVal = $card.find(classIndivdual).val() || "";
        switch (dataIndividual) {
            case "quota":  //額度
                let quotaList = new Array;
                $thisbody.eq(0).find('.quota:checked').map(function () {
                    let number = $.trim($(this).attr('value'));
                    quotaList.push(number);
                });
                individualObj[dataIndividual] = quotaList;
                break;
            case "delay_interest":  //延遲利息計收費方式
                let modeList = new Array;
                $thisbody.eq(0).find('.delay_interest:checked').map(function () {
                    let number = $.trim($(this).attr('value'));
                    modeList.push(number);
                });
                individualObj["mode"] = modeList;
                break;
            case "identities":  //發票人 共同發票人 (暫存 將頁面的塞入)
                let identitiesList = new Array;
                $.each($thisbody.find(".identitiesAllDiv div"), (index, tab) => {
                    let tmpIdentitiesObj = {};
                    tmpIdentitiesObj.code = $(tab).find(".identitiesNo").val();
                    tmpIdentitiesObj.type = $(tab).find(".identitiesRoleSelect").val();
                    tmpIdentitiesObj.name = $(tab).find(".identitiesSelect").val();
                    identitiesList.push(tmpIdentitiesObj);
                })
                individualObj[dataIndividual] = identitiesList;
                break;
            default:
                individualObj[dataIndividual] = ckUnd(classVal);
        }
    });
    return individualObj;
}
