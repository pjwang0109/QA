// 共用：身分別連動綜合契約書下的負責人部分(判斷ID是否符合法人身分) function renderResponsibleList() { let count = {}; let idtArr = []; let $container =$(".responsibleDiv"); // 1. 先暫存目前畫面上已有的負責人資料 (初次載入時這裡為空) let existingIDData = {}; $container.find(".responsible").each(function () { let id = $(this).find(".code").val(); if (id) { existingIDData[id] = { responsibleName: $(this).find(".responsibleName").val() || "", isExpanded: !$(this).find(".principalDiv").hasClass("hide") // 紀錄是否已展開 }; } }); // 2. 清空舊畫面 $container.empty(); // 3. 蒐集符合的身分別資料 $(".identities").each(function () { let $item =$(this); let typeVal = $item.find(".type").val(); let typeText = $item.find(".type option:selected").text(); let name = $item.find(".name").val(); let id = $item.find(".id").val(); if (typeVal) { idtArr.push({ "typeVal": typeVal, "name": name, "id": id, "typeText": typeText }); } }); idtArr.sort((a, b) => a.typeVal - b.typeVal); // 4. 重新渲染輸出 idtArr.forEach((item, index) => { let num = index + 1; let type = item.typeVal; let name = item.name; let id = item.id; let typeText = item.typeText; count[typeText] = (count[typeText] || 0) + 1; if (type) { // 判斷是否符合法人身分 (統編 8 碼) if (id && id.length === 8) { if (type === '0') { typeText = "立約人(即借款人)"; count[typeText] = ""; } // 取得暫存資料（若無暫存則給預設值） let saved = existingIDData[id] || {}; let oldResName = saved.responsibleName || ""; let isExpanded = saved.isExpanded || false; let divClass = isExpanded ? "principalDiv" : "principalDiv hide"; let $responsibleDiv = `
${typeText}${count[typeText]}：
負責人： (自行輸入)
`; $container.append($responsibleDiv); } } }); }
