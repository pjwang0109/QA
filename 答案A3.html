function gu_isClause(obj) { // 只有在當前這顆 radio 被勾選，且值為 "2"（未定期間保證）時才執行清空 if ($(obj).is(":checked") && $(obj).val() === "2") { // 往上找到當前保證人的最外層大容器 let $currentBlock = $(obj).closest(".guarantor-block"); // 也可以加上備用選取器，確保如果沒有 .guarantor-block 時用 parent() 往上找 if ($currentBlock.length === 0) { $currentBlock = $(obj).closest("[id^='guarantor_block_']"); } if ($currentBlock.length === 0) { $currentBlock = $(obj).parent().parent(); // 回退機制 } // 只清空「當前保證人」的年、月、日欄位 $currentBlock.find(".year, .month, .day").val(""); } }
