function isClauseType(obj) { let selectedVal = $(obj).val(); // "1" 或 "2" let $gcc = $('#GCC'); if (selectedVal === "1") { // 1. 取得「主選項」目前填寫的年月日 // 提示：若母欄位的 input 在同一個區塊，可用 $(obj).closest(...) 或指定的 ID/class 取得 let mainYear = $(obj).closest('.form-group').find('.year').val() || $("#mainYear").val() || ""; let mainMonth = $(obj).closest('.form-group').find('.month').val() || $("#mainMonth").val() || ""; let mainDay = $(obj).closest('.form-group').find('.day').val() || $("#mainDay").val() || ""; // 2. 連動所有保證人 $gcc.find("input[type='radio'][name^='clauseType'][value='1']").prop("checked", true); // 3. 日期全部同步填入一樣的值 $gcc.find(".year").val(mainYear); $gcc.find(".month").val(mainMonth); $gcc.find(".day").val(mainDay); } else if (selectedVal === "2") { // 1. 連動所有保證人選取「未定期間保證」 $gcc.find("input[type='radio'][name^='clauseType'][value='2']").prop("checked", true); // 2. 切換為未定期間時，通常需要將所有保證人的年月日清空 $gcc.find(".year").val(""); $gcc.find(".month").val(""); $gcc.find(".day").val(""); } }
