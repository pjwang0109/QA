else if (caseTypeValue == "13") {
    // 撥貸
    $frame.find("#appropriationLoanDiv").removeClass("hide");

    // 不動產地籍異動即時通
    let $messageText = $frame.find("#appropriationLoanDiv").find('.messageText');
    let $messageCG = $frame.find("#appropriationLoanDiv").find('.messageCG');
    
    let hasCollateral = false; // 紀錄是否有擔保品資料

    if (jsondata.result && jsondata.result.collateral) {
        let collateralList = jsondata.result.collateral;
        if (collateralList.length !== 0) {
            hasCollateral = true;
            $messageText.removeClass('hide');
            $messageCG.removeClass('hide');
            $messageCG.empty();

            collateralList.forEach(function (item) {
                if(!item) return;
                let typeVal = item.type;
                let nameVal = item.name;
                let codeVal = item.code;

                let isYesChecked = (nameVal == '1') ? "checked" : "";
                let isNoChecked = (nameVal == '2') ? "checked" : "";
                let isFreeChecked = (nameVal == '3') ? "checked" : "";

                let $rowHtml = `
                    <div class="form-group form-inline collateralRE">
                        <span class="questionStyleA numC ml-4">${typeVal}：</span>
                        <input type="hidden" class="codeVal" value="${codeVal}"></input>
                        <label class="form-check-label ml-2">
                            <input type="radio" class="form-check-input collateral" value="1" name="collateral_${typeVal}" ${isYesChecked} />是
                        </label>
                        <label class="form-check-label ml-2">
                            <input type="radio" class="form-check-input collateral" value="2" name="collateral_${typeVal}" ${isNoChecked} />否
                        </label>
                        <label class="form-check-label ml-2">
                            <input type="radio" class="form-check-input collateral" value="3" name="collateral_${typeVal}" ${isFreeChecked} />免
                        </label>
                        <span>(擔保品非不動產)</span>
                    </div>`;
                $messageCG.append($rowHtml);
            });
        }
    }

    if (!hasCollateral) {
        $messageText.addClass('hide');
        $messageCG.addClass('hide');
        $messageCG.empty();
    }

    // =========================================================================
    // 🌟 1. 先把 $frame 貼到真實網頁畫面上 (假設你的主容器叫 #mainContainer 或彈窗 Modal)
    // =========================================================================
    $('#opCaseDiv').html($frame); // 👈 這裡請替換成你原本專案把 $frame 塞入畫面的那行程式碼！

    // =========================================================================
    // 🌟 2. 確定已經掛載到真實 DOM 後，才呼叫 Tooltip 產生驚嘆號與綁定提示彈窗
    // =========================================================================
    if (hasCollateral) {
        let titleMT = "依消金部115.2.13聯消金字第0031號函規定，於辦理不動產相關業務時，應積極向客戶推廣內政部「地籍異動即時通」服務。";
        
        // 此時從真實 DOM (例如 $('#opCaseDiv')) 裡面去找元素綁定，就絕對會成功！
        methodToolTip.publicToolTip($('#opCaseDiv').find('.messageText .questionStyleA span'), titleMT);
    }
}
