//模板快取（存 ajax 物件，同一支模板只抓一次）
const icTemplateCache = {};
let icTmpSeq = 0;//建立卡片時的暫時編號，建立完會由 renumberIndividualContract 重排

function icGetTemplate(name) {
    if (!icTemplateCache[name]) {
        icTemplateCache[name] = $.ajax(ajaxhttpIPHostHtml(IC_TEMPLATE_PATH + name + ".html"))
            .fail(function () {
                delete icTemplateCache[name];//失敗不快取，下次重抓
            });
    }
    return icTemplateCache[name];
}

//取得類別群組容器，第一次使用時依對照表順序建立全部群組（同類別卡片放在一起）
function icGetGroup(icType) {
    const $tableDiv = $("#indivdualCtableDiv");
    if ($tableDiv.children(".icGroup").length == 0) {
        $.each(individualContractConfig, function (type) {
            $tableDiv.append($('<div class="icGroup"></div>').attr("data-ictype", type));
        });
    }
    return $tableDiv.children(`.icGroup[data-ictype="${icType}"]`);
}

/**
 * 建立一張個別契約卡片並 append 到該類別群組
 * @param {string} icType 類別
 * @returns {jQuery} 卡片元素（之後的 init、回填都對這個元素操作）
 */
async function icCreateCard(icType) {
    const res = await Promise.all([
        icGetTemplate("ic_shell"),
        icGetTemplate("ic_" + icType)
    ]);
    const tmpNum = "tmp" + (++icTmpSeq);
    //字串取代只套用在外框，類別內容 HTML 原樣放入
    const shellHtml = res[0].replace(/demo_/g, '').replace(/Numtitle/g, '').replace(/Num/g, tmpNum);
    const $card = $("<div>").html(shellHtml).find(".indivdualC_TableList").first();
    $card.attr("data-ictype", icType);
    $card.find(".card-body").first().append(res[1]);
    icGetGroup(icType).append($card);
    return $card;
}

//依畫面順序重新編號：id、data-num、collapse 相關屬性為全域連號，標題序號依類別各自計算
function renumberIndividualContract() {
    const seqByType = {};
    //直接對元素改屬性，不用 id 查找（重排過程中可能暫時有重複 id）
    $("#indivdualCtableDiv .indivdualC_TableList").each(function (i) {
        const num = i + 1;
        const $card = $(this);
        let icType = $card.attr("data-ictype");
        //沒有類別的卡片（舊程式建立的）一律視為本票及授權書，並補上類別
        if (!icType) {
            icType = "CASHIER";
            $card.attr("data-ictype", icType);
        }
        const conf = individualContractConfig[icType];
        seqByType[icType] = (seqByType[icType] || 0) + 1;

        const $header = $card.children(".card").children(".card-header");
        const $section = $card.children(".card").children(".collapse");

        $card.attr({ "id": "indivdualCtable" + num, "data-num": num });
        $header.attr("id", "indivdualCheadertable" + num);
        $header.find("h5 a").attr({
            "data-parent": "#indivdualCtable" + num,
            "href": "#indivdualCsectiontable" + num,
            "aria-controls": "indivdualCsectiontable" + num
        }).text((conf ? conf.name : "") + seqByType[icType]);
        $section.attr({
            "id": "indivdualCsectiontable" + num,
            "aria-labelledby": "indivdualCheadertable" + num
        });
    });
}

//約據-個別契約-新增：開啟類別選擇 MODAL
let icSelection = {};//MODAL 的選取狀態 { icType: 份數 }，畫面一律由這個物件產生（切換大項或搜尋時列會重建，選取不會漏掉）

//MODAL-產生左欄大項並綁定事件；右欄的類別列由 icRenderRows 依目前大項或搜尋字產生
function icBuildModal($modal) {
    $modal.find(".icMaxQty").text(IC_MAX_QTY);
    $modal.find(".icMaxTotal").text(IC_MAX_TOTAL);

    const $catList = $modal.find(".icCatList");
    $.each(individualContractCategories, function (i, cat) {
        $catList.append($('<button type="button" class="icCat"></button>')
            .attr("data-category", cat.key)
            .toggleClass("icCatEmpty", icCategoryTypes(cat.key).length == 0)
            .append($('<span class="icCatName"></span>').text(cat.no + "、" + cat.name))
            .append('<span class="icCatQty d-none"></span>'));
    });
    $modal.data("category", individualContractCategories[0].key);

    //整列點選：未選 → 選 1 份；已選 → 取消
    $modal.on("click", ".icRow", function (e) {
        if ($(e.target).closest(".icCtl").length > 0) { return; }
        const icType = icRowType(this);
        icSetQty(icType, icSelection[icType] > 0 ? 0 : 1);
    });
    $modal.on("click", ".icAddBtn", function () {
        icSetQty(icRowType(this), 1);
    });
    $modal.on("click", ".icPlus", function () {
        const icType = icRowType(this);
        icSetQty(icType, (icSelection[icType] || 0) + 1);
    });
    $modal.on("click", ".icMinus", function () {
        const icType = icRowType(this);
        icSetQty(icType, (icSelection[icType] || 0) - 1);
    });
    //份數輸入：只允許數字；輸入中清空或 0 先不處理，離開欄位才生效（避免輸入到一半列被取消選取）
    $modal.on("input", ".icQtyInput", function () {
        digitalboxHandler(this);
        if (this.value === "" || Number(this.value) === 0) { return; }
        icSetQty(icRowType(this), this.value);
    });
    $modal.on("change", ".icQtyInput", function () {
        icSetQty(icRowType(this), this.value);
    });
    $modal.on("click", ".icChipRemove", function () {
        icSetQty($(this).closest(".icChip").attr("data-ictype"), 0);
    });
    $modal.on("click", ".icClearAll", function () {
        icSelection = {};
        icRefresh("");
    });
    //點大項：切換右欄內容，並清掉搜尋
    $modal.on("click", ".icCat", function () {
        $modal.data("category", $(this).attr("data-category"));
        $modal.find(".icSearchInput").val("");
        icRenderRows($modal);
    });
    $modal.on("input", ".icSearchInput", function () {
        icRenderRows($modal);
    });
    $modal.on("click", ".icSearchClear", function () {
        $modal.find(".icSearchInput").val("").trigger("focus");
        icRenderRows($modal);
    });
    //搜尋框按 Enter：只剩一筆符合時直接選 1 份
    $modal.on("keydown", ".icSearchInput", function (e) {
        if (e.key !== "Enter") { return; }
        e.preventDefault();
        const $hits = $modal.find(".icRow");
        if ($hits.length === 1) {
            const icType = $hits.attr("data-ictype");
            if (!(icSelection[icType] > 0)) { icSetQty(icType, 1); }
            this.select();
        }
    });

    icRenderRows($modal);
    icRefresh("", $modal);
}

//取得屬於某大項的類別（依對照表順序）
function icCategoryTypes(categoryKey) {
    return $.grep(Object.keys(individualContractConfig), function (icType) {
        return (individualContractConfig[icType].categories || []).indexOf(categoryKey) >= 0;
    });
}

//取得類別所屬大項的顯示名稱，搜尋結果標示用，例：基本/常用約據、其他
function icCategoryNames(conf) {
    return $.map($.grep(individualContractCategories, function (cat) {
        return (conf.categories || []).indexOf(cat.key) >= 0;
    }), function (cat) { return cat.name; }).join("、");
}

//MODAL-建立單一類別列（未選：＋按鈕；已選：− 份數 ＋）；categoryNames 有值時顯示在名稱下方（搜尋結果用）
function icBuildRow(icType, conf, categoryNames) {
    const $row = $('<div class="icRow"></div>').attr("data-ictype", icType);
    const $text = $('<span class="icRowText"></span>').append('<span class="icName"></span>');
    if (categoryNames) {
        $text.append($('<span class="icRowCats"></span>').text(categoryNames));
    }
    $row.append($text);
    const $ctl = $('<span class="icCtl"></span>');
    $ctl.append('<button type="button" class="icAddBtn" aria-label="新增"><i class="fa fa-plus" aria-hidden="true"></i></button>');
    $ctl.append($('<span class="icStepper"></span>')
        .append('<button type="button" class="icMinus" aria-label="減少"><i class="fa fa-minus" aria-hidden="true"></i></button>')
        .append('<input type="text" class="icQtyInput" maxlength="2" inputmode="numeric" aria-label="份數">')
        .append('<button type="button" class="icPlus" aria-label="增加"><i class="fa fa-plus" aria-hidden="true"></i></button>'));
    $row.append($ctl);
    return $row;
}

function icRowType(el) {
    return $(el).closest(".icRow").attr("data-ictype");
}

//全形轉半形（含括號）、轉小寫、去空白，讓「切結書（不動產抵押）」也能用半形括號搜尋
function icNormalize(str) {
    return String(str || "").normalize("NFKC").toLowerCase().replace(/\s+/g, "");
}

function icTotal() {
    let total = 0;
    $.each(icSelection, function (icType, qty) { total += qty; });
    return total;
}

//MODAL-設定某類別的份數：每類上限 IC_MAX_QTY、單次總數上限 IC_MAX_TOTAL，超過時自動壓回上限並提示
function icSetQty(icType, qty) {
    qty = parseInt(qty, 10) || 0;
    const others = icTotal() - (icSelection[icType] || 0);
    let notice = "";
    if (qty > IC_MAX_QTY) {
        qty = IC_MAX_QTY;
        notice = "每類最多 " + IC_MAX_QTY + " 份";
    }
    if (others + qty > IC_MAX_TOTAL) {
        qty = IC_MAX_TOTAL - others;
        notice = "單次最多新增 " + IC_MAX_TOTAL + " 份";
    }
    if (qty > 0) {
        icSelection[icType] = qty;
    } else {
        delete icSelection[icType];
    }
    icRefresh(notice);
}

//MODAL-依 icSelection 更新類別列的選取狀態與份數
function icSyncRows($modal) {
    $modal.find(".icRow").each(function () {
        const $row = $(this);
        const qty = icSelection[$row.attr("data-ictype")] || 0;
        $row.toggleClass("icSel", qty > 0);
        const $input = $row.find(".icQtyInput");
        if (!($input.is(":focus") && $input.val() === "")) {
            $input.val(qty > 0 ? qty : "");
        }
    });
}

//MODAL-依 icSelection 更新畫面：類別列、左欄各大項份數、已選清單、摘要、確認按鈕
function icRefresh(notice, $modal) {
    $modal = $modal || $("#icTypeModal");
    icSyncRows($modal);
    $modal.find(".icCat").each(function () {
        let qty = 0;
        $.each(icCategoryTypes($(this).attr("data-category")), function (i, icType) {
            qty += icSelection[icType] || 0;
        });
        $(this).find(".icCatQty").text(qty).toggleClass("d-none", qty == 0);
    });

    const $chips = $modal.find(".icChips").empty();
    let count = 0;
    $.each(individualContractConfig, function (icType, conf) {
        const qty = icSelection[icType];
        if (!(qty > 0)) { return; }
        count++;
        $chips.append($('<span class="icChip"></span>').attr("data-ictype", icType)
            .append($('<span class="icChipText"></span>').text(conf.name + " ").append($('<em class="icChipQty"></em>').text("×" + qty)))
            .append('<button type="button" class="icChipRemove" aria-label="移除">&times;</button>'));
    });
    $modal.find(".icTray").toggleClass("d-none", count == 0);

    const total = icTotal();
    $modal.find(".icSummary").text(count == 0 ? "尚未選擇類別" : "已選 " + count + " 類・共 " + total + " 份");
    $modal.find(".icConfirmBtn").text(count == 0 ? "確認新增" : "新增 " + total + " 份");
    $modal.find(".icNotice").text(notice || "");
}

//MODAL-產生右欄類別列：有搜尋字時查全部大項（符合的文字標色，並標示所屬大項），沒有時列出目前大項的類別
function icRenderRows($modal) {
    const rawText = $modal.find(".icSearchInput").val();
    const q = icNormalize(rawText);
    const category = $modal.data("category");
    $modal.find(".icSearchClear").toggleClass("d-none", !rawText);

    const icTypes = q == "" ? icCategoryTypes(category) : $.grep(Object.keys(individualContractConfig), function (icType) {
        const conf = individualContractConfig[icType];
        //搜尋比對：名稱與別名（全形半形、大小寫、空白一律正規化）
        return $.grep([conf.name].concat(conf.keywords || []), function (key) {
            return icNormalize(key).indexOf(q) >= 0;
        }).length > 0;
    });

    const $rows = $modal.find(".icRows").empty();
    $.each(icTypes, function (i, icType) {
        const conf = individualContractConfig[icType];
        const $row = icBuildRow(icType, conf, q == "" ? "" : icCategoryNames(conf));
        icHighlight($row.find(".icName"), conf.name, q);
        $rows.append($row);
    });

    $modal.find(".icCat").each(function () {
        const active = q == "" && $(this).attr("data-category") == category;
        $(this).toggleClass("icCatOn", active);
        if (active) {
            $(this).attr("aria-current", "true");
        } else {
            $(this).removeAttr("aria-current");
        }
    });
    const current = $.grep(individualContractCategories, function (cat) { return cat.key == category; })[0];
    $modal.find(".icLabelCategory").text(current.name).toggleClass("d-none", q != "");
    $modal.find(".icResultInfo").text("全部大項，符合「" + rawText.trim() + "」共 " + icTypes.length + " 筆")
        .toggleClass("d-none", q == "");
    $modal.find(".icEmpty").text(q == "" ? "此大項尚未開放" : "找不到符合的類別")
        .toggleClass("d-none", icTypes.length > 0);
    $modal.find(".icListBox").scrollTop(0);
    icSyncRows($modal);
}

//MODAL-名稱中符合搜尋字的部分標色（正規化後長度不同時不標色，避免位置錯位）
function icHighlight($span, name, q) {
    $span.empty();
    const normalized = name.normalize("NFKC").toLowerCase();
    const idx = (q && normalized.length === name.length) ? normalized.indexOf(q) : -1;
    if (idx < 0) {
        $span.text(name);
        return;
    }
    $span.append(
        document.createTextNode(name.slice(0, idx)),
        $('<mark class="icMark"></mark>').text(name.slice(idx, idx + q.length)),
        document.createTextNode(name.slice(idx + q.length))
    );
}

//MODAL-確認新增：依對照表順序、各類別數量追加卡片
async function addIndividualContractBatch(target) {
    const selections = [];
    $.each(individualContractConfig, function (icType) {//依對照表順序
        const qty = icSelection[icType];
        if (qty > 0) {
            selections.push({ icType: icType, qty: Math.min(qty, IC_MAX_QTY) });
        }
    });
    if (selections.length == 0) {
        alert("請勾選要新增的類別");
        return;
    }
    let total = 0;
    $.each(selections, function (i, sel) { total += sel.qty; });
    if (total > IC_MAX_TOTAL) {
        alert("單次最多新增 " + IC_MAX_TOTAL + " 份");
        return;
    }

    $(target).prop("disabled", true);
    try {

/**
 * S060000005(約據傳送) 上行維持舊格式：individualContracts 裡 icType 為 CASHIER 的本票轉回 cashiers
 * @param {Object} data saveOpJSON 的結果
 * @returns {Object} 新物件，不影響傳入的 data
 */
function icToLegacyCashiers(data) {
    const result = $.extend({}, data);
    result.cashiers = $.map(data.individualContracts || [], function (item) {
        if (item.icType != "CASHIER") { return null; }
        //no 排第一、seal1 放回 seal2 前面，與原本 cashiers 的欄位順序相同
        const legacy = { "no": item.no };
        $.each(item.data, function (key, value) {
            if (key == "seal2") { legacy.seal1 = item.seal1; }
            legacy[key] = value;
        });
        if (!("seal1" in legacy)) { legacy.seal1 = item.seal1; }
        return legacy;
    });
    delete result.individualContracts;
    return result;
}

/**
 * 約據-個別契約-依下行資料（S060001036）重建卡片
 * @param {Array} individualContracts 個別契約 [{ no, icType, seal1, index, data }]
 * @param {boolean} isDisabled 是否鎖住卡片
 */
async function renderIndividualContract(individualContracts, isDisabled) {
    const items = [];
    $.each(individualContracts || [], function (i, val) {
        if (!individualContractConfig[val.icType]) {
            console.error("個別契約：未定義的類別 icType=" + val.icType + "，略過不顯示");
            return;
        }
        items.push({ icType: val.icType, no: val.no, seal1: val.seal1 || "", index: icParseIndex(val.index), pos: i, data: val.data || {} });
    });
    if (items.length == 0) { return; }
    //同一類別內依 index 排序；index 相同依原陣列順序，缺少 index 的排在最後（維持原陣列順序）
    //（不同類別各自放在自己的分組容器，跨類別的先後不影響畫面）
    items.sort(function (a, b) {
        const aHas = a.index !== null, bHas = b.index !== null;
        if (aHas !== bHas) { return aHas ? -1 : 1; }
        if (aHas && a.index !== b.index) { return a.index - b.index; }
        return a.pos - b.pos;
    });

    try {
        //依序建立，確保畫面順序與資料順序一致
        for (const item of items) {
            item.$card = await icCreateCard(item.icType);
        }
        renumberIndividualContract();
        $.each(items, function (i, item) {
            icRunInit(item.$card);
        });
        icAfterRender();//回填利率型式前，選單要先有選項

        $.each(items, function (i, item) {
            item.$card.find(".no").first().val(item.no);//編號（外層）
            item.$card.find(".seal1").first().val(item.seal1);//簽章（外層）
            const conf = individualContractConfig[item.icType];
            if (conf && typeof conf.fill === "function") {
                conf.fill(item.$card, item.data);
            }
            if (isDisabled) {
                item.$card.find("input, select, button").prop("disabled", true);
            }
        });
        icRefreshPageFields();//fill 只記住所選人員的統編，由這裡對到身分別並帶入欄位

        $("#indivdualC .indivdualCsection").collapse('show');
        $.each(items, function (i, item) {
            icShowCard(item.$card);
        });
    } catch (e) {
        console.error(e);
    }
}

//下行 index 轉成數字；沒有或不是數字回傳 null
function icParseIndex(value) {
    if (value === undefined || value === null || value === "") { return null; }
    const num = Number(value);
    return isFinite(num) ? num : null;
}

/**
 * 個別契約-共用欄位：類別的內容 HTML 放上這些 class，就會從頁面帶值並自動連動；沒放的就跳過，類別不用寫程式
 *
 * 人員：包在 .icPersonGroup 內，一張卡片可以有多組
 *   data-types：限定身分別代碼，如 "6" 或 "0,6"；不寫則列出全部
 *   .icPerson 下拉選單、.icPersonName 名稱、.icPersonResponsible 負責人、.icPersonAddress 住址、.icPersonId 統編/ID
 *   .icPersonIdLabel、.icPersonDateLabel 依 ID 長度切換的標題文字
 * 借款人：.icBorrowerName 借款人名稱（固定抓身分別代碼 0 那一筆，不用下拉，也不用包在 .icPersonGroup 內；跟下拉一樣，沒填 ID 視為還沒有這個人）
 * 其他共同條款：.icCourts 約定管轄法院、.icPhone 分行電話、.icFax 分行傳真
 * 民國日期：包在 .icDateGroup 內，.icDateYear、.icDateMonth、.icDateDay
 *   放在 .icPersonGroup 內時視為這個人的日期，換人會清空；放在外面則不受影響
 *
 * 類別的 fill、collect 用 icSetPerson、icGetPerson、icSetDate、icGetDate 存取
 * ※class 一律用 ic 開頭，避免被 $(".address")、$(".courts") 這類整頁選擇器誤讀誤寫
 */
const IC_DATE_YEAR_MAX = 150;
const IC_DATE_SELECTOR = "#indivdualCtableDiv .icDateYear, #indivdualCtableDiv .icDateMonth, #indivdualCtableDiv .icDateDay";

//人員-回填：記住統編，由 icRefreshPageFields 對到身分別（groupIndex 為卡片內第幾組人員，預設第一組）
function icSetPerson($card, id, groupIndex) {
    const $group = $card.find(".icPersonGroup").eq(groupIndex || 0);
    $group.data("icPersonId", $.trim(id || ""));
    $group.data("icPersonRow", null);
}

//人員-取值：回傳所選身分別的統編/ID，先依頁面現況刷新一次，確保是最新的
function icGetPerson($card, groupIndex) {
    const $group = $card.find(".icPersonGroup").eq(groupIndex || 0);
    icRefreshPersonGroup($group, icIdentityList());
    return $group.data("icPersonId") || "";
}

//民國日期-回填：照下行原樣帶入（不檢查上限），val 為 { year, month, day }
function icSetDate($card, val, groupIndex) {
    const $group = $card.find(".icDateGroup").eq(groupIndex || 0);
    icWriteDate($group.find(".icDateYear"), val.year || "");
    icWriteDate($group.find(".icDateMonth"), val.month || "");
    icWriteDate($group.find(".icDateDay"), val.day || "");
}

//民國日期-取值：回傳 { year, month, day }
function icGetDate($card, groupIndex) {
    const $group = $card.find(".icDateGroup").eq(groupIndex || 0);
    return {
        "year": $group.find(".icDateYear").val() || "",
        "month": $group.find(".icDateMonth").val() || "",
        "day": $group.find(".icDateDay").val() || ""
    };
}

/**
 * 依身分別區塊（#genderDiv .identities）的畫面順序，列出有身分別且有統編/ID 的人員
 * 選項文字為身分別名稱，同一種身分別有多筆時加序號（連帶保證人1、連帶保證人2）
 * @returns {Array} [{ el: 身分別元素, type, id, label, name, address, responsible }]
 */
function icIdentityList() {
    const list = [];
    $("#genderDiv .identities").each(function () {
        const $row = $(this);
        const $type = $row.find("select.type").first();
        const id = $.trim($row.find("input.id").first().val() || "");
        if (!$type.val() || id == "") { return; }
        list.push({
            el: this,
            type: $type.val(),
            id: id,
            typeText: $.trim($type.find("option:selected").text()),
            name: $.trim($row.find("input.name").first().val() || ""),
            address: icIdentityAddress($row),
            responsible: icIdentityResponsible(id)
        });
    });
    const total = {}, seq = {};
    $.each(list, function (i, item) { total[item.typeText] = (total[item.typeText] || 0) + 1; });
    $.each(list, function (i, item) {
        seq[item.typeText] = (seq[item.typeText] || 0) + 1;
        item.label = item.typeText + (total[item.typeText] > 1 ? seq[item.typeText] : "");
    });
    return list;
}

/**
 * 身分別地址組成一行
 * 中華民國：郵遞區號 縣市區路 巷 弄 號(之) 樓 其他，例：310002 新竹縣竹東鎮大明路8巷9弄12之1號5樓
 * 其他：郵遞區號 國家 地址；都沒勾選回傳空字串
 * ※縣市、區、路、國家是 selectpicker，外框 div 也會帶相同 class，要用 select. 指定
 */
function icIdentityAddress($row) {
    const v = function (selector) { return $.trim($row.find(selector).first().val() || ""); };
    const flag = $row.find(".address-flag:checked").val();
    if (flag == "1") {
        const of = v("input.of"), sub = v("input.num");//of:號、num:之
        let no = "";
        if (of && sub) {
            no = of + "之" + sub + "號";
        } else if (of || sub) {
            no = (of || sub) + "號";
        }
        const lane = v("input.lane"), aly = v("input.aly"), flr = v("input.flr");
        const body = v("select.city") + v("select.area") + v("select.street")
            + (lane ? lane + "巷" : "") + (aly ? aly + "弄" : "") + no
            + (flr ? flr + "樓" : "") + v("input.etc");
        return $.grep([v("input.zip"), body], function (s) { return s != ""; }).join(" ");
    }
    if (flag == "2") {
        const $nation = $row.find("select.nationO").first();
        const nation = $nation.val() ? $.trim($nation.find("option:selected").text()) : "";
        return $.grep([v("input.addZip"), nation, v("input.add")], function (s) { return s != ""; }).join(" ");
    }
    return "";
}

//負責人：responsibleDiv 中 .code 等於該統編的那一列（只有 8 碼統編才會有），未展開「負責人」時為空白
function icIdentityResponsible(id) {
    let name = "";
    $(".responsibleDiv .responsible").each(function () {
        if ($.trim($(this).find(".code").val() || "") !== id) { return; }
        if (!$(this).find(".principalDiv").hasClass("hide")) {
            name = $.trim($(this).find(".responsibleName").val() || "");
        }
        return false;
    });
    return name;
}

/**
 * 刷新一組人員：重建下拉選項並帶入欄位
 * 這組人員記住所選的身分別元素（icPersonRow）與統編（icPersonId）：元素還在就跟著它（身分別的 ID 被修改也不會斷）；
 * 元素不在了再用統編找；都找不到（身分別被刪除）就回到「請選擇」
 */
function icRefreshPersonGroup($group, list) {
    const typesAttr = $group.attr("data-types");
    const types = typesAttr ? typesAttr.split(",") : null;
    const options = types ? $.grep(list, function (item) { return types.indexOf(item.type) >= 0; }) : list;

    const row = $group.data("icPersonRow");
    const id = $group.data("icPersonId");
    let idx = -1;
    if (row) {
        idx = options.findIndex(function (item) { return item.el === row; });
    }
    if (idx < 0 && id) {
        idx = options.findIndex(function (item) { return item.id === id; });
    }
    const match = idx >= 0 ? options[idx] : null;
    $group.data("icPersonRow", match ? match.el : null);
    $group.data("icPersonId", match ? match.id : "");

    //選項有變才重建，避免使用者展開選單時被重繪
    const $select = $group.find(".icPerson").first();
    const signature = $.map(options, function (item) { return item.id + "\u0001" + item.label; }).join("\u0002");
    if ($select.data("icSignature") !== signature) {
        $select.find("option").not(":first").remove();
        $.each(options, function (i, item) {
            $select.append($("<option></option>").val(item.id).text(item.label));
        });
        $select.data("icSignature", signature);
    }
    $select.data("icOptions", options);
    $select.prop("selectedIndex", idx + 1);//用位置選取，同一統編出現在兩筆身分別時也能選到正確那筆

    const idLength = match ? match.id.length : 0;
    $group.find(".icPersonIdLabel").text(idLength == 8 ? "營利事業統一編號：" : idLength == 10 ? "國民身分證統一編號：" : "營利事業統一編號／國民身分證統一編號：");
    $group.find(".icPersonDateLabel").text(idLength == 8 ? "設立日期：" : idLength == 10 ? "出生年月日：" : "設立日期／出生年月日：");
    $group.find(".icPersonName").val(match ? match.name : "");
    $group.find(".icPersonResponsible").val(match ? match.responsible : "");
    $group.find(".icPersonAddress").val(match ? match.address : "");
    $group.find(".icPersonId").val(match ? match.id : "");
}

//刷新所有卡片的共用欄位（沒有用到共用欄位時不做事）
function icRefreshPageFields() {
    const $tableDiv = $("#indivdualCtableDiv");
    const $groups = $tableDiv.find(".icPersonGroup");
    const $fields = $tableDiv.find(".icBorrowerName, .icCourts, .icPhone, .icFax");
    if ($groups.length == 0 && $fields.length == 0) { return; }
    icWatchPageFields();
    const list = icIdentityList();
    $groups.each(function () {
        icRefreshPersonGroup($(this), list);
    });
    const borrower = list.find(function (item) { return item.type == "0"; });
    $tableDiv.find(".icBorrowerName").val(borrower ? borrower.name : "");
    const $source = $("#otherCommonTermsDiv");
    $tableDiv.find(".icCourts").val($source.find(".courts").first().val() || "");
    $tableDiv.find(".icPhone").val($source.find(".phone").first().val() || "");
    $tableDiv.find(".icFax").val($source.find(".fax").first().val() || "");
}

let icRefreshTimer = null;
function icScheduleRefresh() {
    clearTimeout(icRefreshTimer);
    icRefreshTimer = setTimeout(icRefreshPageFields, 100);
}

/**
 * 監看身分別與負責人區塊的 DOM 變動
 * 程式用 .val() 帶值不會觸發事件，而讀取時身分別的地址是非同步帶入（agreementData 等 API 回來才填區、路），
 * 帶完最後會重繪 selectpicker；新增/刪除身分別、renderResponsibleList 重建負責人也都會改到 DOM，
 * 所以監看這兩個區塊的變動，延遲 0.1 秒刷新一次
 */
let icObserver = null;
let icObserved = [];
function icWatchPageFields() {
    if (!window.MutationObserver) { return; }
    const targets = $("#genderDiv, .responsibleDiv").get();
    const same = icObserver && targets.length === icObserved.length
        && targets.every(function (el, i) { return el === icObserved[i]; });
    if (same) { return; }
    if (icObserver) { icObserver.disconnect(); }
    icObserver = new MutationObserver(icScheduleRefresh);
    $.each(targets, function (i, el) {
        icObserver.observe(el, { childList: true, subtree: true, characterData: true, attributes: true, attributeFilter: ["class"] });
    });
    icObserved = targets;
}

//使用者換人：記住所選的身分別；放在這組人員內的日期是這個人的資料，一併清空
function icOnPersonChange() {
    const $group = $(this).closest(".icPersonGroup");
    const options = $(this).data("icOptions") || [];
    const item = options[this.selectedIndex - 1] || null;
    $group.data("icPersonRow", item ? item.el : null);
    $group.data("icPersonId", item ? item.id : "");
    icWriteDate($group.find(".icDateYear, .icDateMonth, .icDateDay"), "");
    icRefreshPageFields();
}

/**
 * 民國日期的輸入上限，只檢查使用者的輸入，程式帶入的值（S060001036）照原樣顯示
 * 年 1～150、月 1～12、日 1～該年月的最後一天；輸入 0 清空；超過上限擋住（退回輸入前的值）
 * 改年或月後，已填的日超過新的上限 → 改成該月最後一天
 */
//日的上限：月不合法（空白或超出 1～12）→ 31；年不合法 → 無法判斷閏年，2 月給 29；都合法 → 依實際年月（民國年 + 1911）
function icDateMaxDay($group) {
    const year = Number($group.find(".icDateYear").val());
    const month = Number($group.find(".icDateMonth").val());
    if (!(month >= 1 && month <= 12)) { return 31; }
    if (!(year >= 1 && year <= IC_DATE_YEAR_MAX)) { return month == 2 ? 29 : new Date(2001, month, 0).getDate(); }
    return new Date(year + 1911, month, 0).getDate();
}

//程式設定年月日時一併記錄「輸入前的值」，之後使用者輸入超過上限才能退回正確的值
function icWriteDate($input, value) {
    $input.val(value);
    $input.each(function () { $(this).data("icDateLast", this.value); });
}

//年月日-進入欄位時記錄目前的值（超過上限時退回用）
function icOnDateFocus() {
    $(this).data("icDateLast", this.value);
}

//年月日-輸入時檢查：去掉非數字與開頭的 0、0 清空、超過上限退回輸入前的值
function icOnDateInput() {
    const $input = $(this);
    const digits = this.value.replace(/\D/g, "");
    let value = digits === "" ? "" : String(Number(digits));//去掉開頭的 0
    if (value === "0") { value = ""; }
    const max = $input.hasClass("icDateYear") ? IC_DATE_YEAR_MAX : $input.hasClass("icDateMonth") ? 12 : icDateMaxDay($input.closest(".icDateGroup"));
    if (value !== "" && Number(value) > max) {
        const last = $input.data("icDateLast");
        value = last === undefined ? "" : last;
    }
    this.value = value;
    $input.data("icDateLast", value);
}

//改完年或月（離開欄位）：日超過新的上限就改成該月最後一天（輸入年份途中，如 1→11→113，不會誤改日）
function icOnYearMonthChange() {
    const $group = $(this).closest(".icDateGroup");
    const $day = $group.find(".icDateDay");
    const max = icDateMaxDay($group);
    if ($day.val() !== "" && Number($day.val()) > max) {
        icWriteDate($day, String(max));
    }
}

/**
 * 共用欄位的事件綁定（檔案載入時執行一次）
 * 用事件委派綁在 document：卡片是之後才建立的、身分別與負責人區塊會被重建，綁在 document 都照樣有效
 * ※不能放在 init：init 每張卡片都會跑，會在 document 上疊加多個相同的處理函式，卡片刪除後也不會消失
 */
function icBindPageFieldEvents() {

    $(document).on("input change", "#genderDiv input, #genderDiv select, .responsibleDiv .responsibleName, #otherCommonTermsDiv .courts, #otherCommonTermsDiv .phone, #otherCommonTermsDiv .fax", icScheduleRefresh);
    //使用者換人
    $(document).on("change", "#indivdualCtableDiv .icPerson", icOnPersonChange);
    //民國日期
    $(document).on("focusin", IC_DATE_SELECTOR, icOnDateFocus);
    $(document).on("input", IC_DATE_SELECTOR, icOnDateInput);
    $(document).on("change", "#indivdualCtableDiv .icDateYear, #indivdualCtableDiv .icDateMonth", icOnYearMonthChange);
}
icBindPageFieldEvents();

//本票及授權書-回填（val 為下行的 data）
function icCashiersFill($card, val) {
    const num = $card.attr("data-num");
    let $identitiesAllDiv = $card.find(".identitiesAllDiv");

    //發票人或共同發票人開始
    $.each(val.identities, function (i, v) {
        if (i > 0) {
            //判斷如果大於一筆時
            i++;
            let $div = `<div class="mt-2 identitiesSelectDiv${i}">
            <input type="hidden" class="identitiesNo" name="no" value="${v.code}">
            <select class="form-control mr-2 identitiesRoleSelect">
                <option value="">請選擇</option>
                <option value="1">發票人</option>
                <option value="2">共同發票人</option>
            </select>
            <select class="form-control mr-2 identitiesSelect">
                <option value="">請選擇</option>
            </select>
            <button type="button" class="btn btn-danger mr-2" onclick="delIdentitiescBtn(this)">×</button>
            </div>`;

    let quota_Class = $card.find(".quota");

//本票及授權書-暫存取值（S060001037 的 data）
//由 saveOpJSON 原本的本票取值搬來，欄位順序 quota, amount, type, mode, interest, other, seal2, identities 不可變動（S060000005 轉回 cashiers 時沿用此順序）
//data 內 quota:額度Type, amount:金額, type:利率型式, mode:延遲利息計收方式(複選), interest:加碼年息, other:其他, seal2:授權書簽章, identities:發票人
function icCashiersCollect($card) {
    const classIndividualArr = [".quota", ".amount", ".type", ".delay_interest", ".interest", ".other", ".interest", ".seal2", ".identities"];
    const $thisbody = $card.find('.card-body');
    const individualObj = new Object;
    $.each(classIndividualArr, function (item, classIndivdual) {
        const dataIndividual = classIndivdual.replace('.', '');
        const classVal = $card.find(classIndivdual).val() || "";
        switch (dataIndividual) {
            case "quota":  //額度
                let quotaList = new Array;
                $thisbody.eq(0).find('.quota:checked').map(function () {
                    let number = $.trim($(this).attr('value'));
                    quotaList.push(number);
                });
                individualObj[dataIndividual] = quotaList;
                break;
            case "delay_interest":  //延遲利息計收費方式
                let modeList = new Array;
                $thisbody.eq(0).find('.delay_interest:checked').map(function () {
                    let number = $.trim($(this).attr('value'));
                    modeList.push(number);
                });
                individualObj["mode"] = modeList;
                break;
            case "identities":  //發票人 共同發票人 (暫存 將頁面的塞入)
                let identitiesList = new Array;
                $.each($thisbody.find(".identitiesAllDiv div"), (index, tab) => {
                    let tmpIdentitiesObj = {};
                    tmpIdentitiesObj.code = $(tab).find(".identitiesNo").val();
                    tmpIdentitiesObj.type = $(tab).find(".identitiesRoleSelect").val();
                    tmpIdentitiesObj.name = $(tab).find(".identitiesSelect").val();
                    identitiesList.push(tmpIdentitiesObj);
                })
                individualObj[dataIndividual] = identitiesList;
                break;
            default:
                individualObj[dataIndividual] = ckUnd(classVal);
        }
    });
    return individualObj;
}
