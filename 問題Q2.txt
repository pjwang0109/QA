/**
 * 刷新一組人員：重建下拉選項並帶入欄位
 * 這組人員記住所選的身分別元素（icPersonRow）與統編（icPersonId）：元素還在就跟著它（身分別的 ID 被修改也不會斷）；
 * 元素不在了再用統編找；都找不到（身分別被刪除）就回到「請選擇」
 */

function icRefreshPersonGroup($group, list) {
    const typesAttr = $group.attr("data-types");
    const types = typesAttr ? typesAttr.split(",") : null;
    const options = types ? $.grep(list, function (item) { return types.indexOf(item.type) >= 0; }) : list;

    const row = $group.data("icPersonRow");
    const id = $group.data("icPersonId");
    let idx = -1;
    if (row) {
        idx = options.findIndex(function (item) { return item.el === row; });
    }
    if (idx < 0 && id) {
        idx = options.findIndex(function (item) { return item.id === id; });
    }
    const match = idx >= 0 ? options[idx] : null;
    $group.data("icPersonRow", match ? match.el : null);
    $group.data("icPersonId", match ? match.id : "");

    //選項有變才重建，避免使用者展開選單時被重繪
    const $select = $group.find(".icPerson").first();
    const signature = $.map(options, function (item) { return item.id + "\u0001" + item.label; }).join("\u0002");
    if ($select.data("icSignature") !== signature) {
        $select.find("option").not(":first").remove();
        $.each(options, function (i, item) {
            $select.append($("<option></option>").val(item.id).text(item.label));
        });
        $select.data("icSignature", signature);
    }
    $select.data("icOptions", options);
    $select.prop("selectedIndex", idx + 1);//用位置選取，同一統編出現在兩筆身分別時也能選到正確那筆

    const idLength = match ? match.id.length : 0;
    $group.find(".icPersonIdLabel").text(idLength == 8 ? "營利事業統一編號：" : idLength == 10 ? "國民身分證統一編號：" : "營利事業統一編號／國民身分證統一編號：");
    $group.find(".icPersonDateLabel").text(idLength == 8 ? "設立日期：" : idLength == 10 ? "出生年月日：" : "設立日期／出生年月日：");
    $group.find(".icPersonName").val(match ? match.name : "");
    $group.find(".icPersonResponsible").val(match ? match.responsible : "");
    $group.find(".icPersonAddress").val(match ? match.address : "");
    $group.find(".icPersonId").val(match ? match.id : "");
}
