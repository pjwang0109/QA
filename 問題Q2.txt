$($target).on(
        "click",
        ".import401Div .table .del-btn[data-num]",
        (event) => {
            if (confirm("是否刪除?")) {
                const $btn = $(event.target);
                const $table = $btn.closest("table");
                const delNum = $btn.data("num") || "1";
                if (delNum == "1") {
                    $btn.closest("tr").remove();
                } else if (delNum == "2") {
                    const nextTr = $btn.closest("tr").next("tr");
                    nextTr.remove();
                    $btn.closest("tr").remove();
                }
                $.each($table.find(".del-btn[data-row]"), (i, btnObj) => {
                    $(btnObj).attr("data-row", i + 1);
                });
            }
        }
    );
