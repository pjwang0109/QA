function handlePhoneTimeKeydown(e, element) { // 檢查是否按下 Enter 鍵 if (e.key === "Enter" || e.keyCode === 13) { e.preventDefault(); // 防止預設送出行為 let $input = $(element); let val = $input.val().trim(); // 判斷是否為 4 碼純數字 (例如 "1430" 或 "0905") if (/^\d{4}$/.test(val)) { let hour = parseInt(val.substring(0, 2), 10); let minute = parseInt(val.substring(2, 4), 10); // 驗證時間範圍合法性 (00~23 時、00~59 分) if (hour >= 0 && hour <= 23 && minute >= 0 && minute <= 59) { let formattedTime = `${val.substring(0, 2)}:${val.substring(2, 4)}`; // 若有綁定 Flatpickr，透過實例更新能確保內部同步 if (element._flatpickr) { element._flatpickr.setDate(formattedTime, true); element._flatpickr.close(); } else { $input.val(formattedTime).trigger('change'); } $input.blur(); } else { alert("時間超出合法範圍（時需為 00-23，分需為 00-59）"); } } } }
